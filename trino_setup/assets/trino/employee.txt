    CAST(e.primary_country_code AS varchar) AS primary_country_code
FROM {{SHARD_CATALOG}}.application.reporting_employee e
WHERE e.client_code = split_part(CURRENT_USER, '|', 2)
  AND EXISTS (
      SELECT 1
      FROM {{SHARD_CATALOG}}.application.employee_access a
      WHERE a.clientcode = split_part(CURRENT_USER, '|', 2)
        AND a.username = split_part(CURRENT_USER, '|', 3)
        AND a.clientcode = e.client_code
        AND a.eecode = e.employee_code
  );
